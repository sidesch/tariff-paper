/*******************************************************************************
Clean trade import data from Data Web
**********************************************************************************/
clear

global data "~/Desktop/school/6th/ecn374/finaldraft/data"
cap mkdir "$data/processed"

import excel "$data/raw/DataWeb-Query-Export.xlsx", sheet("Query Results") firstrow
drop DataType Country Description
rename January_2017_to_december_2017 customs_value
rename HTSNumber hts10

drop if missing(hts10)
gen hs6 = real(substr(hts10,1,6))
destring customs_value, replace
collapse (sum) customs_value, by(hs6)

save "$data/processed/imports_clean.dta", replace

* Excerpt from /code/cleaning/02_clean_concordance.do
/*********************************************************************************
Clean INEGI's concordance between TIGIE (Mexican tariff schedule) and SCIAN
(NAICS) values.
************************************************************************************/
clear
set more off

global data "~/Desktop/school/6th/ecn374/finaldraft/data"

import excel "$data/raw/concordance_inegi.xlsx", cellrange(A5:E10000) firstrow
rename A tigie_code
rename D scian
cap tostring scian, replace
drop Reproductores~a Explotaciónde~s C

drop if missing(tigie_code)
replace tigie_code = subinstr(tigie_code, ".", "", .)  
gen hs6 = real(substr(tigie_code, 1, 6))      
gen scian_num = real(scian)

* merge with hs6 import data
merge m:1 hs6 using "$data/processed/imports_clean.dta", keep(3)
drop _merge

save "$data/processed/tigie_scian_clean.dta", replace
